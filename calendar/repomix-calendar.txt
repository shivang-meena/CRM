// apps/client/src/calendar/CalendarScreen.tsx
import React, { useMemo, useState } from 'react';
import { ActivityIndicator, StyleSheet, Text, useWindowDimensions, View } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { errorMessage } from '@/api/client';
import Button from '@/components/Button';
import { colors } from '@/theme/colors';
import { useCalendarActions, useCalendarItems } from '@/hooks/useCalendar';
import { CALENDAR_CATEGORIES, CalendarCategory, CalendarEventInput, CalendarItem, CalendarView } from '@/types/calendar';
import CalendarFilters from './CalendarFilters';
import CalendarToolbar from './CalendarToolbar';
import MonthCalendar from './MonthCalendar';
import DayCalendar from './DayCalendar';
import CalendarList from './CalendarList';
import CalendarEventFormModal from './CalendarEventFormModal';
import CalendarItemDetailsModal from './CalendarItemDetailsModal';
import { addDays, addMonths, formatDayHeading, formatMonthHeading, todayIstKey, visibleRange } from './calendarDates';

const COMPACT_BREAKPOINT = 720;
const DANGER_SOFT = '#FFF1F0';

export default function CalendarScreen() {
  const { width } = useWindowDimensions();
  const compact = width < COMPACT_BREAKPOINT;

  const [view, setView] = useState<CalendarView>('month');
  const [selectedDate, setSelectedDate] = useState(todayIstKey());
  const [selectedCategories, setSelectedCategories] = useState<CalendarCategory[]>([...CALENDAR_CATEGORIES]);
  const [formOpen, setFormOpen] = useState(false);
  const [editingEvent, setEditingEvent] = useState<CalendarItem | null>(null);
  const [detailItem, setDetailItem] = useState<CalendarItem | null>(null);

  const range = useMemo(() => visibleRange(view, selectedDate), [view, selectedDate]);
  const query = useMemo(() => ({ from: range.from, to: range.to, categories: selectedCategories }), [range.from, range.to, selectedCategories]);
  const itemsQuery = useCalendarItems(query);
  const actions = useCalendarActions();
  const heading = view === 'day' ? formatDayHeading(selectedDate) : formatMonthHeading(selectedDate);

  function toggleCategory(category: CalendarCategory) {
    setSelectedCategories(current => current.includes(category) ? current.filter(item => item !== category) : [...current, category]);
  }
  function navigate(amount: number) {
    setSelectedDate(current => view === 'day' ? addDays(current, amount) : addMonths(current, amount));
  }
  function openAdd() { setEditingEvent(null); setFormOpen(true); }
  function openEdit(item: CalendarItem) { setDetailItem(null); setEditingEvent(item); setFormOpen(true); }
  async function saveEvent(data: CalendarEventInput) {
    if (editingEvent) await actions.update.mutateAsync({ id: editingEvent.sourceId, data });
    else await actions.create.mutateAsync(data);
    setFormOpen(false);
    setEditingEvent(null);
  }
  async function deleteEvent(item: CalendarItem) {
    if (!item.version) return;
    await actions.remove.mutateAsync({ id: item.sourceId, version: item.version });
  }

  const items = itemsQuery.data?.items ?? [];

  return (
    <View style={styles.container}>
      <View style={[styles.layout, compact ? styles.layoutCompact : styles.layoutWide]}>
        <CalendarFilters layout={compact ? 'bar' : 'sidebar'} selected={selectedCategories} onToggle={toggleCategory} onAdd={openAdd} />

        <View style={[styles.main, !compact && styles.mainCard]}>
          <CalendarToolbar
            heading={heading}
            view={view}
            onView={setView}
            onPrevious={() => navigate(-1)}
            onNext={() => navigate(1)}
            onToday={() => setSelectedDate(todayIstKey())}
          />

          {itemsQuery.isLoading && !itemsQuery.data ? (
            <View style={styles.state}>
              <ActivityIndicator color={colors.accent} />
              <Text style={styles.stateText}>Loading calendar…</Text>
            </View>
          ) : itemsQuery.isError ? (
            <View accessibilityRole="alert" style={styles.state}>
              <View style={styles.errorIcon}><Ionicons name="alert-circle-outline" size={26} color={colors.danger} /></View>
              <Text style={styles.stateTitle}>Couldn't load the calendar</Text>
              <Text style={styles.error}>{errorMessage(itemsQuery.error)}</Text>
              <Button title="Retry" variant="outline" onPress={() => { void itemsQuery.refetch(); }} />
            </View>
          ) : (
            <>
              {!items.length ? (
                <View style={styles.notice}>
                  <Ionicons name="information-circle-outline" size={16} color={colors.muted} />
                  <Text style={styles.noticeText}>No entries match the selected filters.</Text>
                </View>
              ) : null}
              {view === 'month' ? (
                <MonthCalendar
                  month={selectedDate}
                  items={items}
                  selectedDate={selectedDate}
                  onSelectDate={date => { setSelectedDate(date); setView('day'); }}
                  onMore={date => { setSelectedDate(date); setView('day'); }}
                  onOpenItem={setDetailItem}
                />
              ) : view === 'day' ? (
                <DayCalendar date={selectedDate} items={items} onOpenItem={setDetailItem} />
              ) : (
                <CalendarList items={items} onOpenItem={setDetailItem} />
              )}
            </>
          )}
        </View>
      </View>

      <CalendarEventFormModal
        visible={formOpen}
        selectedDate={selectedDate}
        event={editingEvent}
        onClose={() => { if (!actions.create.isPending && !actions.update.isPending) { setFormOpen(false); setEditingEvent(null); } }}
        onSave={saveEvent}
      />
      <CalendarItemDetailsModal item={detailItem} onClose={() => setDetailItem(null)} onEdit={openEdit} onDelete={deleteEvent} />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: colors.canvas },
  layout: { flex: 1 },
  layoutWide: { flexDirection: 'row', flexWrap: 'nowrap', alignItems: 'stretch', padding: 16, gap: 16 },
  layoutCompact: { flexDirection: 'column' },
  main: { flex: 1, minWidth: 0, minHeight: 0, backgroundColor: colors.white },
  mainCard: { borderWidth: 1, borderColor: colors.border, borderRadius: 16, overflow: 'hidden', boxShadow: '0 4px 14px rgba(11,47,125,0.07)' },
  state: { flex: 1, minHeight: 260, alignItems: 'center', justifyContent: 'center', gap: 10, padding: 24 },
  stateText: { color: colors.muted, fontSize: 13 },
  stateTitle: { color: colors.text, fontSize: 16, fontWeight: '800' },
  errorIcon: { width: 52, height: 52, borderRadius: 26, backgroundColor: DANGER_SOFT, alignItems: 'center', justifyContent: 'center' },
  error: { color: colors.danger, fontSize: 13, textAlign: 'center', marginBottom: 4 },
  notice: { flexDirection: 'row', alignItems: 'center', gap: 6, marginHorizontal: 14, marginTop: 10, paddingHorizontal: 10, paddingVertical: 7, borderRadius: 8, backgroundColor: '#F7F9FC', alignSelf: 'flex-start' },
  noticeText: { color: colors.muted, fontSize: 12 },
});

import React, { useState } from 'react';
import { Pressable, StyleSheet, Text, View } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { colors } from '@/theme/colors';
import { CalendarView } from '@/types/calendar';
import { IconName, SOFT } from './calendarUi';

type Props = { heading: string; view: CalendarView; onView: (view: CalendarView) => void; onPrevious: () => void; onNext: () => void; onToday: () => void };

const VIEW_OPTIONS: { key: CalendarView; icon: IconName }[] = [
  { key: 'month', icon: 'grid-outline' },
  { key: 'day', icon: 'today-outline' },
  { key: 'list', icon: 'list-outline' },
];

export default function CalendarToolbar({ heading, view, onView, onPrevious, onNext, onToday }: Props) {
  const [width, setWidth] = useState(0);
  const compact = width > 0 && width < 640;

  const arrows = (
    <View style={styles.arrows}>
      <Pressable accessibilityRole="button" accessibilityLabel="Previous calendar period" onPress={onPrevious} style={({ pressed }) => [styles.arrow, pressed && styles.pressed]}>
        <Ionicons name="chevron-back" size={18} color={colors.text} />
      </Pressable>
      <View style={styles.arrowDivider} />
      <Pressable accessibilityRole="button" accessibilityLabel="Next calendar period" onPress={onNext} style={({ pressed }) => [styles.arrow, pressed && styles.pressed]}>
        <Ionicons name="chevron-forward" size={18} color={colors.text} />
      </Pressable>
    </View>
  );

  const todayButton = (
    <Pressable accessibilityRole="button" accessibilityLabel="Go to today" onPress={onToday} style={({ pressed }) => [styles.today, pressed && styles.pressed]}>
      <Text style={styles.todayText}>Today</Text>
    </Pressable>
  );

  const views = (
    <View style={[styles.views, compact && styles.viewsCompact]}>
      {VIEW_OPTIONS.map(({ key, icon }) => {
        const active = view === key;
        return (
          <Pressable
            key={key}
            accessibilityRole="button"
            accessibilityState={{ selected: active }}
            onPress={() => onView(key)}
            style={[styles.viewButton, compact && styles.viewButtonCompact, active && styles.viewButtonActive]}
          >
            <Ionicons name={icon} size={14} color={active ? colors.primaryText : colors.muted} />
            <Text style={[styles.viewText, active && styles.viewTextActive]}>{key[0].toUpperCase() + key.slice(1)}</Text>
          </Pressable>
        );
      })}
    </View>
  );

  return (
    <View style={styles.toolbar} onLayout={(event) => setWidth(event.nativeEvent.layout.width)}>
      {compact ? (
        <>
          <View style={styles.row}>
            <Text accessibilityRole="header" numberOfLines={1} style={[styles.heading, styles.headingFlex]}>{heading}</Text>
            {todayButton}
          </View>
          <View style={styles.row}>
            {arrows}
            {views}
          </View>
        </>
      ) : (
        <View style={styles.row}>
          {arrows}
          <Text accessibilityRole="header" numberOfLines={1} style={[styles.heading, styles.headingFlex]}>{heading}</Text>
          {todayButton}
          {views}
        </View>
      )}
    </View>
  );
}

// apps/client/src/calendar/MonthCalendar.tsx
import React, { useState } from 'react';
import { Pressable, ScrollView, StyleSheet, Text, useWindowDimensions, View } from 'react-native';
import { colors } from '@/theme/colors';
import { CALENDAR_CATEGORY_COLORS, CalendarItem } from '@/types/calendar';
import { firstDayOfMonth, formatCalendarTime, monthGridKeys, parseDateKey, overlapsIstDay, todayIstKey } from './calendarDates';
import { TODAY_TINT, tint } from './calendarUi';

type Props = { month: string; items: CalendarItem[]; selectedDate: string; onSelectDate: (date: string) => void; onOpenItem: (item: CalendarItem) => void; onMore: (date: string) => void };

const WEEKDAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
const HEADER_HEIGHT = 40;
const MIN_CELL_HEIGHT = 72;
const DOT_MODE_WIDTH = 64; // cells narrower than this (phones) show dots instead of text chips
const TIME_IN_CHIP_WIDTH = 112;

export default function MonthCalendar({ month, items, selectedDate, onSelectDate, onOpenItem, onMore }: Props) {
  const { height: viewportHeight } = useWindowDimensions();
  const [size, setSize] = useState({ width: 0, height: 0 });

  const today = todayIstKey();
  const first = firstDayOfMonth(month);
  const days = monthGridKeys(month);
  const rowCount = days.length / 7;

  // Cells fill the space that is really available; the old window-based estimate is only a first-paint fallback.
  const available = size.height > 0
    ? Math.floor((size.height - HEADER_HEIGHT) / rowCount)
    : Math.floor((viewportHeight - 170) / rowCount);
  const cellHeight = Math.max(MIN_CELL_HEIGHT, available);
  const cellWidth = size.width > 0 ? size.width / 7 : 100;
  const dotMode = cellWidth < DOT_MODE_WIDTH;
  const showTime = cellWidth >= TIME_IN_CHIP_WIDTH;
  const maxChips = dotMode ? 3 : Math.max(1, Math.min(4, Math.floor((cellHeight - 36 - 16) / 22)));

  return (
    <View style={styles.calendar} onLayout={(event) => setSize(event.nativeEvent.layout)}>
      <View style={styles.weekHeader}>
        {WEEKDAYS.map(day => <Text key={day} style={styles.weekDay}>{day}</Text>)}
      </View>

      <ScrollView style={styles.scroll} showsVerticalScrollIndicator={false}>
        <View style={styles.grid}>
          {days.map(date => {
            const dateItems = items
              .filter(item => overlapsIstDay(item, date))
              .sort((a, b) => new Date(a.startAt).getTime() - new Date(b.startAt).getTime() || a.key.localeCompare(b.key));
            const visibleItems = dateItems.slice(0, maxChips);
            const hiddenCount = dateItems.length - visibleItems.length;
            const { day } = parseDateKey(date);
            const outside = !date.startsWith(first.slice(0, 7));
            const isToday = date === today;

import React from 'react';
import { Pressable, ScrollView, StyleSheet, Text, View } from 'react-native';

import { CategoryTag, EmptyState, IconName, SOFT, StatusTag, tint } from './calendarUi';

type Props = { date: string; items: CalendarItem[]; onOpenItem: (item: CalendarItem) => void };

export default function DayCalendar({ date, items, onOpenItem }: Props) {
  const entries = items.filter(item => overlapsIstDay(item, date));
  const allDay = entries.filter(item => item.allDay);
  const timed = entries
    .filter(item => !item.allDay)
    .sort((a, b) => new Date(a.startAt).getTime() - new Date(b.startAt).getTime());

  if (!entries.length) {
    return (
      <ScrollView contentContainerStyle={styles.body}>
        <EmptyState icon="calendar-clear-outline" title="Nothing scheduled" text="No entries for this day." />
      </ScrollView>
    );
  }

  return (
    <ScrollView contentContainerStyle={styles.body}>
      {allDay.length ? (
        <View style={styles.section}>
          <SectionTitle icon="sunny-outline" label="All day" count={allDay.length} />
          {allDay.map(item => <DayEntry key={item.key} item={item} onOpen={onOpenItem} />)}
        </View>
      ) : null}

      {timed.length ? (
        <View style={styles.section}>
          <SectionTitle icon="time-outline" label="Timed" count={timed.length} />
          {timed.map(item => <DayEntry key={item.key} item={item} date={date} onOpen={onOpenItem} />)}
        </View>
      ) : null}
    </ScrollView>
  );
}

function SectionTitle({ icon, label, count }: { icon: IconName; label: string; count: number }) {
  return (
    <View style={styles.sectionHeader}>
      <Ionicons name={icon} size={15} color={colors.muted} />
      <Text style={styles.sectionTitle}>{label}</Text>
      <View style={styles.countPill}><Text style={styles.countText}>{count}</Text></View>
    </View>
  );
}

function DayEntry({ item, date, onOpen }: { item: CalendarItem; date?: string; onOpen: (item: CalendarItem) => void }) {
  const color = CALENDAR_CATEGORY_COLORS[item.category];

  // Timed entries get a time rail; entries that cross midnight say so instead of showing a misleading time.
  let railStart = '';
  let railEnd = '';
  if (date && !item.allDay) {
    railStart = todayIstKey(new Date(item.startAt)) < date ? 'Ongoing' : formatCalendarTime(item.startAt);
    if (item.endAt) railEnd = todayIstKey(new Date(item.endAt)) > date ? 'Continues' : formatCalendarTime(item.endAt);
  }

  return (
    <View style={styles.entryRow}>
      {date ? (
        <View style={styles.rail}>
          <Text style={styles.railStart}>{railStart}</Text>
          {railEnd ? <Text style={styles.railEnd}>{railEnd}</Text> : null}
        </View>
      ) : null}
      <Pressable
        accessibilityRole="button"
        accessibilityLabel={item.title}
        onPress={() => onOpen(item)}
        style={({ pressed }) => [styles.entry, { backgroundColor: tint(color, pressed ? '26' : '14'), borderLeftColor: color }]}
      >
        <Text style={styles.title}>{item.title}</Text>
        <View style={styles.tags}>
          <CategoryTag category={item.category} />
          {item.status ? <StatusTag status={item.status} /> : null}
        </View>
      </Pressable>
    </View>
  );
}

type Props = { items: CalendarItem[]; onOpenItem: (item: CalendarItem) => void };

const WEEKDAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

// One readable "when" line per entry (the old version repeated the time and the range).
function whenLabel(item: CalendarItem): string {
  if (item.allDay) {
    if (!item.endAt) return 'All day';
    const startKey = todayIstKey(new Date(item.startAt));
    const endKey = addDays(todayIstKey(new Date(item.endAt)), -1);
    return startKey === endKey ? 'All day' : itemRangeLabel(item);
  }
  if (!item.endAt) return formatCalendarTime(item.startAt);
  return todayIstKey(new Date(item.startAt)) === todayIstKey(new Date(item.endAt))
    ? `${formatCalendarTime(item.startAt)} – ${formatCalendarTime(item.endAt)}`
    : itemRangeLabel(item);
}

export default function CalendarList({ items, onOpenItem }: Props) {
  const groups = useMemo(() => {
    const map = new Map<string, CalendarItem[]>();
    [...items]
      .sort((a, b) => new Date(a.startAt).getTime() - new Date(b.startAt).getTime() || a.key.localeCompare(b.key))
      .forEach(item => {
        const key = todayIstKey(new Date(item.startAt));
        map.set(key, [...(map.get(key) ?? []), item]);
      });
    return [...map.entries()];
  }, [items]);

  if (!groups.length) {
    return (
      <ScrollView contentContainerStyle={styles.body}>
        <EmptyState icon="calendar-clear-outline" title="No calendar entries" text="There are no entries for this month." />
      </ScrollView>
    );
  }

  const today = todayIstKey();

  return (
    <ScrollView contentContainerStyle={styles.body}>
      {groups.map(([date, rows]) => {
        const { day, month } = parseDateKey(date);
        const weekday = WEEKDAYS[dateKeyToUtc(date).getUTCDay()];
        const isToday = date === today;
        return (
          <View key={date} style={styles.group}>
            <View
              accessible
              accessibilityLabel={`${weekday} ${day} ${MONTHS[month - 1]}`}
              style={[styles.badge, isToday && styles.badgeToday]}
            >
              <Text style={[styles.badgeWeekday, isToday && styles.badgeTodayText]}>{weekday}</Text>
              <Text style={[styles.badgeDay, isToday && styles.badgeTodayText]}>{day}</Text>
              <Text style={[styles.badgeMonth, isToday && styles.badgeTodayText]}>{MONTHS[month - 1]}</Text>
            </View>

            <View style={styles.items}>
              {rows.map(item => (
                <Pressable
                  key={item.key}
                  accessibilityRole="button"
                  accessibilityLabel={item.title}
                  onPress={() => onOpenItem(item)}
                  style={({ pressed }) => [styles.card, pressed && styles.cardPressed]}
                >
                  <View style={[styles.bar, { backgroundColor: CALENDAR_CATEGORY_COLORS[item.category] }]} />
                  <View style={styles.main}>
                    <Text style={styles.title}>{item.title}</Text>
                    <View style={styles.whenRow}>
                      <Ionicons name="time-outline" size={13} color={colors.muted} />
                      <Text style={styles.when}>{whenLabel(item)}</Text>
                    </View>
                    <View style={styles.tags}>

}

import React, { useEffect, useState } from 'react';
import { Modal, Pressable, ScrollView, StyleSheet, Text, TextInput, View } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import Button from '@/components/Button';
import DateTimePickerModal from '@/components/DateTimePickerModal';
import { errorMessage } from '@/api/client';
import { colors } from '@/theme/colors';
import { CALENDAR_CATEGORY_COLORS, CalendarEventInput, CalendarItem } from '@/types/calendar';
import { addDays, dateKeyToIstMidnightIso, dateTimeKeyToIstIso, formatCalendarTime, formatDateKey, parseDateKey, todayIstKey } from './calendarDates';
import { SURFACE, tint } from './calendarUi';

type Props = { visible: boolean; selectedDate: string; event: CalendarItem | null; onClose: () => void; onSave: (data: CalendarEventInput) => Promise<void> };
type FormState = { title: string; category: CalendarEventInput['category']; description: string; allDay: boolean; startDate: string; endDate: string; startTime: string; endTime: string };

const DANGER_SOFT = '#FFF1F0';
const requestKey = () => `cal_${Date.now().toString(36)}_${Math.random().toString(36).slice(2, 10)}`;
const pickerValue = (date: string, time: string) => { const { year, month, day } = parseDateKey(date); return `${String(day).padStart(2, '0')}-${String(month).padStart(2, '0')}-${year} ${time}`; };
const parsePicker = (value: string) => { const match = /^(\d{2})-(\d{2})-(\d{4}) (\d{2}:\d{2})$/.exec(value); return match ? { date: `${match[3]}-${match[2]}-${match[1]}`, time: match[4] } : null; };
const initial = (selectedDate: string, event: CalendarItem | null): FormState => {
  if (!event) return { title: '', category: 'EVENT', description: '', allDay: false, startDate: selectedDate, endDate: selectedDate, startTime: '10:00', endTime: '11:00' };
  const startDate = todayIstKey(new Date(event.startAt)); const endDate = event.endAt ? todayIstKey(new Date(event.endAt)) : startDate;
  return { title: event.title, category: event.category as CalendarEventInput['category'], description: event.description ?? '', allDay: event.allDay, startDate, endDate: event.allDay ? addDays(endDate, -1) : endDate, startTime: formatCalendarTime(event.startAt), endTime: event.endAt ? formatCalendarTime(event.endAt) : '11:00' };
};

export default function CalendarEventFormModal({ visible, selectedDate, event, onClose, onSave }: Props) {
  const [form, setForm] = useState<FormState>(() => initial(selectedDate, event));
  const [picker, setPicker] = useState<'startDate' | 'endDate' | 'startTime' | 'endTime' | null>(null);
  const [error, setError] = useState('');
  const [saving, setSaving] = useState(false);

  useEffect(() => { if (visible) { setForm(initial(selectedDate, event)); setPicker(null); setError(''); } }, [visible, selectedDate, event]);

  const set = <K extends keyof FormState>(key: K, value: FormState[K]) => setForm(old => ({ ...old, [key]: value }));

  async function save() {
    if (!form.title.trim()) return setError('Title is required.');
    if (form.endDate < form.startDate) return setError('End date must be on or after start date.');
    const startAt = form.allDay ? dateKeyToIstMidnightIso(form.startDate) : dateTimeKeyToIstIso(form.startDate, form.startTime);
    const endAt = form.allDay ? dateKeyToIstMidnightIso(addDays(form.endDate, 1)) : dateTimeKeyToIstIso(form.endDate, form.endTime);
    if (new Date(endAt) <= new Date(startAt)) return setError('End must be after start.');
    setSaving(true); setError('');
    try {
      await onSave({ title: form.title.trim(), category: form.category, description: form.description.trim() || null, startAt, endAt, allDay: form.allDay, timezone: 'Asia/Kolkata', ...(event ? { version: event.version ?? 1 } : { requestKey: requestKey() }) });
      onClose();
    } catch (cause) {
      setError(errorMessage(cause));
    } finally {
      setSaving(false);
    }
  }

  const pickerDate = picker === 'endDate' || picker === 'endTime' ? form.endDate : form.startDate;
  const pickerTime = picker === 'startTime' ? form.startTime : picker === 'endTime' ? form.endTime : '12:00';

  return (
    <Modal visible={visible} transparent animationType="fade" onRequestClose={() => !saving && onClose()}>
      <View style={styles.overlay}>
        <View style={styles.modal}>
          <View style={styles.header}>
            <View style={styles.headerText}>
              <Text style={styles.headerTitle}>{event ? 'Edit Calendar Event' : 'Add Calendar Event'}</Text>
            </View>
            <Pressable accessibilityRole="button" accessibilityLabel="Close event form" disabled={saving} onPress={onClose} hitSlop={8} style={styles.closeButton}>
              <Ionicons name="close" size={20} color={colors.primaryText} />
            </Pressable>
          </View>

          <ScrollView contentContainerStyle={styles.body} keyboardShouldPersistTaps="handled">
            <Field label="Title *" value={form.title} onChangeText={value => set('title', value)} />

            <View style={styles.field}>
              <Text style={styles.label}>Category</Text>
              <View style={styles.chips}>
                {(['EVENT', 'HOLIDAY', 'SERVICE'] as CalendarEventInput['category'][]).map(value => {
                  const active = form.category === value;
                  const color = CALENDAR_CATEGORY_COLORS[value];
                  return (
                    <Pressable
                      key={value}
                      accessibilityRole="button"
                      accessibilityState={{ selected: active }}
                      onPress={() => set('category', value)}
                      style={[styles.chip, active && { backgroundColor: tint(color, '1A'), borderColor: color }]}
                    >
                      <View style={[styles.chipDot, { backgroundColor: color }]} />
                      <Text style={[styles.chipText, active && styles.chipTextActive]}>{value[0] + value.slice(1).toLowerCase()}</Text>
                      {active ? <Ionicons name="checkmark" size={14} color={color} /> : null}
                    </Pressable>
                  );
                })}
              </View>
            </View>

            <Field label="Description" value={form.description} onChangeText={value => set('description', value)} multiline />

            {/* Everything about "when" lives in one panel */}
            <View style={styles.panel}>
              <Pressable accessibilityRole="switch" accessibilityState={{ checked: form.allDay }} onPress={() => set('allDay', !form.allDay)} style={styles.toggleRow}>
                <View style={styles.toggleLabel}>
                  <Ionicons name="sunny-outline" size={17} color={colors.muted} />
                  <Text style={styles.toggleText}>All day</Text>
                </View>
                <View style={[styles.toggle, form.allDay && styles.toggleActive]}><View style={[styles.knob, form.allDay && styles.knobActive]} /></View>
              </Pressable>

              <View style={styles.row}>
                <View style={styles.half}><DateField label="Start date" value={form.startDate} onPress={() => setPicker('startDate')} /></View>
                <View style={styles.half}><DateField label="End date" value={form.endDate} onPress={() => setPicker('endDate')} /></View>
              </View>

              {!form.allDay ? (
                <View style={styles.row}>
                  <View style={styles.half}><DateField label="Start time" value={form.startTime} onPress={() => setPicker('startTime')} time /></View>
                  <View style={styles.half}><DateField label="End time" value={form.endTime} onPress={() => setPicker('endTime')} time /></View>
                </View>
              ) : (
                <View style={styles.helperRow}>
                  <Ionicons name="information-circle-outline" size={15} color={colors.muted} />
                  <Text style={styles.helper}>The end date is inclusive and will be stored as the next day at IST midnight.</Text>
                </View>
              )}
            </View>

            {error ? (
              <View accessibilityRole="alert" style={styles.errorBox}>
                <Ionicons name="alert-circle-outline" size={17} color={colors.danger} />
                <Text style={styles.error}>{error}</Text>
              </View>
            ) : null}
          </ScrollView>

          <View style={styles.footer}>
            <Button title="Cancel" variant="outline" onPress={onClose} disabled={saving} />
            <Button title={event ? 'Save Changes' : 'Save Event'} onPress={() => { void save(); }} loading={saving} />
          </View>
        </View>
      </View>

      <DateTimePickerModal
        visible={!!picker}
        value={picker ? pickerValue(pickerDate, pickerTime) : ''}
        title={picker?.includes('Time') ? 'Choose Time' : picker === 'endDate' ? 'Choose End Date' : 'Choose Start Date'}
        onSelect={value => {
          const parsed = parsePicker(value);
          if (!parsed || !picker) return;
          if (picker === 'startDate' || picker === 'endDate') set(picker, parsed.date); else set(picker, parsed.time);
          setPicker(null);
        }}
        onClose={() => setPicker(null)}
      />
    </Modal>
  );
}

function Field({ label, value, onChangeText, multiline = false }: { label: string; value: string; onChangeText: (value: string) => void; multiline?: boolean }) {

        multiline={multiline}

    </View>
  );

        <Text style={styles.dateText}>{time ? value : formatDateKey(value)}</Text>
      </Pressable>
    </View>
  );
}

export default function CalendarItemDetailsModal({ item, onClose, onEdit, onDelete }: Props) {
  const router = useRouter();
  const [confirm, setConfirm] = useState(false);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');
  if (!item) return null;
  const currentItem = item;
  const color = CALENDAR_CATEGORY_COLORS[currentItem.category];
  const sourceLabel = currentItem.source === 'CALENDAR_EVENT' ? 'Calendar event' : currentItem.source === 'LEAD_FOLLOWUP' ? 'Lead follow-up' : currentItem.source === 'REMINDER' ? 'Reminder' : 'Meeting';

  function openSource() {
    if (currentItem.leadId) {
      router.push({ pathname: '/lead/[id]', params: { id: currentItem.leadId, section: currentItem.source === 'LEAD_FOLLOWUP' ? 'followups' : currentItem.source === 'REMINDER' ? 'reminders' : 'meetings' } });
    } else if (currentItem.source === 'REMINDER') router.push('/reminders');
    else if (currentItem.source === 'MEETING') router.push('/meetings');
  }

  async function remove() {
    setBusy(true);
    setError('');
    try {
      await onDelete(currentItem);
      setConfirm(false);
      onClose();
    } catch (cause) {
      setError(errorMessage(cause));
    } finally {
      setBusy(false);
    }
  }

  const when = currentItem.allDay
    ? itemRangeLabel(currentItem)
    : currentItem.endAt
      ? itemRangeLabel(currentItem)
      : `${formatCalendarDate(currentItem.startAt)} ${formatCalendarTime(currentItem.startAt)}`;

  return (
    <>

          <View style={styles.modal}>

              </Pressable>
            </View>

            <ScrollView contentContainerStyle={styles.body}>
              <View style={[styles.hero, { backgroundColor: tint(color, '14'), borderLeftColor: color }]}>
                <Text style={styles.eventTitle}>{currentItem.title}</Text>
                <View style={styles.tags}>
                  <CategoryTag category={currentItem.category} />
                  <View style={styles.sourceTag}><Text style={styles.sourceText}>{sourceLabel}</Text></View>
                </View>
              </View>

              <View style={styles.infoCard}>
                <InfoRow icon="time-outline" label="When" value={when} />
                <View style={styles.rowDivider} />
                <InfoRow icon="flag-outline" label="Status" value={currentItem.status ? prettyStatus(currentItem.status) : '—'} />
                <View style={styles.rowDivider} />
                <InfoRow icon="document-text-outline" label="Description" value={currentItem.description || '—'} />
              </View>

              {error ? (
                <View accessibilityRole="alert" style={styles.errorBox}>
                  <Ionicons name="alert-circle-outline" size={17} color={colors.danger} />
                  <Text style={styles.error}>{error}</Text>
                </View>
              ) : null}
            </ScrollView>

            <View style={styles.footer}>
              {currentItem.editable ? (
                <>
                  <Button title="Edit" variant="outline" onPress={() => onEdit(currentItem)} />
                  <Button title="Delete" variant="danger" onPress={() => setConfirm(true)} />
                </>
              ) : currentItem.source !== 'CALENDAR_EVENT' ? (
                <Button title="Open source" variant="outline" onPress={openSource} />
              ) : null}
              <Button title="Close" onPress={onClose} />
            </View>
          </View>

      </Modal>

      <ConfirmDialog
        visible={confirm}

        message="This event will be archived and removed from the calendar."
        loading={busy}
        onCancel={() => setConfirm(false)}
        onConfirm={() => { void remove(); }}
      />
    </>
  );
}

// apps/client/src/calendar/calendarDates.ts
import { CalendarItem, CalendarView } from '@/types/calendar';

const IST_MINUTES = 330;
const DAY_MS = 86400000;
const pad = (value: number) => String(value).padStart(2, '0');

export function todayIstKey(now = new Date()): string {
  const shifted = new Date(now.getTime() + IST_MINUTES * 60000);
  return `${shifted.getUTCFullYear()}-${pad(shifted.getUTCMonth() + 1)}-${pad(shifted.getUTCDate())}`;
}

export function parseDateKey(key: string): { year: number; month: number; day: number } {
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(key);
  if (!match) throw new Error('Invalid calendar date key.');
  return { year: Number(match[1]), month: Number(match[2]), day: Number(match[3]) };
}

export function dateKeyToUtc(key: string): Date { const { year, month, day } = parseDateKey(key); return new Date(Date.UTC(year, month - 1, day)); }
export function dateKeyToIstMidnightIso(key: string): string { return new Date(dateKeyToUtc(key).getTime() - IST_MINUTES * 60000).toISOString(); }
export function dateTimeKeyToIstIso(key: string, time: string): string { const [hour, minute] = time.split(':').map(Number); return new Date(dateKeyToUtc(key).getTime() + (hour * 60 + minute - IST_MINUTES) * 60000).toISOString(); }
export function addDays(key: string, amount: number): string { const date = new Date(dateKeyToUtc(key).getTime() + amount * DAY_MS); return `${date.getUTCFullYear()}-${pad(date.getUTCMonth() + 1)}-${pad(date.getUTCDate())}`; }
export function addMonths(key: string, amount: number): string { const { year, month, day } = parseDateKey(key); const index = year * 12 + month - 1 + amount; const nextYear = Math.floor(index / 12); const nextMonth = index % 12 + 1; const maxDay = new Date(Date.UTC(nextYear, nextMonth, 0)).getUTCDate(); return `${nextYear}-${pad(nextMonth)}-${pad(Math.min(day, maxDay))}`; }
export function firstDayOfMonth(key: string): string { const { year, month } = parseDateKey(key); return `${year}-${pad(month)}-01`; }
export function lastDayOfMonth(key: string): string { const { year, month } = parseDateKey(key); return `${year}-${pad(month)}-${pad(new Date(Date.UTC(year, month, 0)).getUTCDate())}`; }
export function monthGridKeys(key: string): string[] { const first = firstDayOfMonth(key); const offset = dateKeyToUtc(first).getUTCDay(); const days = dateKeyToUtc(lastDayOfMonth(key)).getUTCDate(); const cells = Math.max(35, Math.ceil((offset + days) / 7) * 7); return Array.from({ length: cells }, (_, index) => addDays(first, index - offset)); }
export function visibleRange(view: CalendarView, selected: string): { from: string; to: string } { if (view === 'day') return { from: dateKeyToIstMidnightIso(selected), to: dateKeyToIstMidnightIso(addDays(selected, 1)) }; if (view === 'list') { const first = firstDayOfMonth(selected); return { from: dateKeyToIstMidnightIso(first), to: dateKeyToIstMidnightIso(addDays(first, dateKeyToUtc(lastDayOfMonth(selected)).getUTCDate())) }; } const grid = monthGridKeys(selected); return { from: dateKeyToIstMidnightIso(grid[0]), to: dateKeyToIstMidnightIso(addDays(grid[grid.length - 1], 1)) }; }
export function overlapsIstDay(item: CalendarItem, key: string): boolean { const from = dateKeyToIstMidnightIso(key); const to = dateKeyToIstMidnightIso(addDays(key, 1)); if (!item.endAt) return item.startAt >= from && item.startAt < to; return item.startAt < to && item.endAt > from; }
export function formatMonthHeading(key: string): string { return new Intl.DateTimeFormat('en-IN', { month: 'long', year: 'numeric', timeZone: 'UTC' }).format(dateKeyToUtc(key)); }
export function formatDayHeading(key: string): string { return new Intl.DateTimeFormat('en-IN', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric', timeZone: 'UTC' }).format(dateKeyToUtc(key)); }
export function formatDateKey(key: string): string { return new Intl.DateTimeFormat('en-IN', { day: '2-digit', month: 'short', year: 'numeric', timeZone: 'UTC' }).format(dateKeyToUtc(key)); }
export function formatCalendarTime(iso: string): string { return new Intl.DateTimeFormat('en-IN', { hour: '2-digit', minute: '2-digit', hourCycle: 'h23', timeZone: 'Asia/Kolkata' }).format(new Date(iso)); }
export function formatCalendarDate(iso: string): string { return new Intl.DateTimeFormat('en-IN', { day: '2-digit', month: 'short', year: 'numeric', timeZone: 'Asia/Kolkata' }).format(new Date(iso)); }
export function itemRangeLabel(item: CalendarItem): string { if (item.allDay && item.endAt) return `${formatDateKey(todayIstKey(new Date(item.startAt)))} – ${formatDateKey(addDays(todayIstKey(new Date(item.endAt)), -1))}`; if (!item.endAt) return formatCalendarTime(item.startAt); return `${formatCalendarDate(item.startAt)} ${formatCalendarTime(item.startAt)} – ${formatCalendarDate(item.endAt)} ${formatCalendarTime(item.endAt)}`; }
